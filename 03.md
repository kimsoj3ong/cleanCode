import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.Optional;
import java.util.UUID;


public class MainRefTest {
    public static void main(String[] args) {
        // 1) 정상
        var svc = new RefactoredOrderService(/* 정책/리포지토리 주입 */);
        var req = new CheckoutRequest("vip001", "A-100", 2, true, "COUPON10");
        Receipt r = svc.checkout(req);
        System.out.println(r); // OK:...

        // 2) 검증 실패
        try {
            svc.checkout(new CheckoutRequest("", "A-100", -1, false, ""));
            System.out.println("FAIL: should have thrown");
        } catch(ValidationException e) {
            System.out.println("PASS: " + e.getMessage());
        }

        // 3) 재고 부족
        try {
            svc.checkout(new CheckoutRequest("u1", "B-100", 99, false, ""));
            System.out.println("FAIL: should have thrown");
        } catch(OutOfStockException e) {
            System.out.println("PASS: OUT_OF_STOCK");
        }
    }
}

record CheckoutRequest(String userId, String itemId, int quantity, boolean isVip, String couponCode) {
    CheckoutRequest {
        userId = Optional.ofNullable(userId).orElse("").trim();
        itemId = Optional.ofNullable(itemId).orElse("").trim();
        couponCode = Optional.ofNullable(couponCode).orElse("").trim();
    }
}

record Receipt(String orderId, String userId, String itemId, int qty, BigDecimal sum) {
    @Override
    public String toString() {
        return "OK : " + orderId + " : " + userId + " : " + itemId + " : " + qty + " : " + sum;
    }
}

class RefactoredOrderService {
    // 가격/배송/할인/재고 관련 상수
    private static final BigDecimal UNIT_A = new BigDecimal("100");
    private static final BigDecimal UNIT_OTHER = new BigDecimal("150");

    private static final BigDecimal SHIP_A = new BigDecimal("500");
    private static final BigDecimal SHIP_OTHER = new BigDecimal("1200");

    private static final int WEIGHT_PER_UNIT = 2;     // kg
    private static final int EXTRA_WEIGHT_LIMIT = 10; // kg
    
    private static final BigDecimal EXTRA_FEE = new BigDecimal("800");

    private static final BigDecimal VIP_DISCOUNT_RATE = new BigDecimal("0.90");
    private static final String COUPON_FIXED_CODE = "COUPON10";
    private static final BigDecimal COUPON_FIXED_AMT = new BigDecimal("10");

    private static final int MAX_PER_ORDER = 20; // 재고량

    public Receipt checkout(CheckoutRequest req) {
        // 로깅(임시)
        System.out.println("[CHECKOUT START] req = " + req);
        
        validate(req);
        boolean isA = isCategoryA(req.itemId());

        BigDecimal itemTotal = calcItemTotal(isA, req.quantity());
        BigDecimal shipping = calcShipping(isA, req.quantity());
        BigDecimal beforeDiscount = itemTotal.add(shipping);

        // VIP/쿠폰 할인
        BigDecimal discounted = applyDiscounts(beforeDiscount, req.isVip(), req.couponCode()).setScale(0, RoundingMode.DOWN);
        
        // 재고 확인
        ensureStock(req.quantity());

        // 저장
        String orderId = saveOrder();
        System.out.println("SAVE : order = " + orderId + " user = " + req.userId() + " item = " + req.itemId() + " qty = " + req.quantity() + " sum = " + discounted);

        // 영수증 조립
        Receipt receipt = new Receipt(orderId, req.userId(), req.itemId(), req.quantity(), discounted);
        
        return receipt;
    }
    
    private void validate(CheckoutRequest req) {
        if (req.userId().isEmpty()) throw new ValidationException("USER_REQUIRED");
        if (req.itemId().isEmpty()) throw new ValidationException("ITEM_REQUIRED");
        if (req.quantity() <= 0) throw new ValidationException("QTY_POSITIVE");
    }

    private boolean isCategoryA(String itemId) { 
        return itemId.startsWith("A"); 
    }

    private BigDecimal calcItemTotal(boolean isA, int qty) {
        // A면 100원, 아니면 150원
        BigDecimal unit = isA ? UNIT_A : UNIT_OTHER;
        return unit.multiply(BigDecimal.valueOf(qty));
    }

    private BigDecimal calcShipping(boolean isA, int qty) {
        // A면 500원, 아니면 1200원
        BigDecimal base = isA ? SHIP_A : SHIP_OTHER;
        int weight = WEIGHT_PER_UNIT * qty;
        BigDecimal extra = (weight > EXTRA_WEIGHT_LIMIT) ? EXTRA_FEE : BigDecimal.ZERO;
        return base.add(extra);
    }

    private BigDecimal applyDiscounts(BigDecimal before, boolean isVip, String coupon) {
        BigDecimal sum = before;

        // VIP/쿠폰 할인
        if (isVip) { 
            // 10% 할인
            sum = sum.multiply(VIP_DISCOUNT_RATE);
        }
        
        if (!coupon.isBlank() && COUPON_FIXED_CODE.equalsIgnoreCase(coupon)) {
            // 10원 고정 차감
            sum = sum.subtract(COUPON_FIXED_AMT);
        }
        return sum;
    }

    private void ensureStock(int qty) {
        if (qty > MAX_PER_ORDER) {
            throw new OutOfStockException("OUT_OF_STOCK");
        }    
    }

    private String saveOrder() {
        return UUID.randomUUID().toString();
    }
}

class ValidationException extends RuntimeException { 
    ValidationException(String m){
        super(m);
    } 
}
class OutOfStockException extends RuntimeException { 
    OutOfStockException(String m){
        super(m);
    } 
}
